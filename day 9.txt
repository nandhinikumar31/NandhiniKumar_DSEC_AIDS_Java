		    );

		int units = Integer.parseInt(br.readLine());
		double bill;
		if(units < 0) {
		    System.out.println("Invalid units");
		    return;
		}
		if(units <= 100) {
		    bill = units * 2;
		} else if(units <= 200) {
		    bill = (100 * 2) + (units - 100) * 3;
		} else {
		    bill = (100 * 2) + (100 * 2) + (units - 100) * 5;
		}
		System.out.println("electricity bill: Rs " + bill);
		

	}
}

		String wordsToRemove = sc.nextLine();
		String[] words = sentence.split(" ");
		StringBuilder sb = new StringBuilder();
		
		for(int i = 0;i < words.length;i++) {
		    if(!words[i].equals(wordsToRemove)) {

		        sb.append(" ");
		    
		    }
		}
		}
	
		
		System.out.print(sb.toString().trim().replaceAll(" +"," "));

	}
}

13.import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner sc = new Scanner(System.in);
		String word = sc.nextLine();
		String prefix = sc.nextLine();
		String suffix = sc.nextLine();
		
		if(word.contains(":")) word = word.split(":")[1].trim();
		if(prefix.contains(":")) prefix = prefix.split(":")[1].trim();
		if(suffix.contains(":")) suffix = suffix.split(":")[1].trim();
		
		if(prefix.contains("suffix")) {
		    String[] parts = prefix.split("suffix:");
		    prefix = parts[0].trim();
		    suffix = parts[1].trim();
		}
		StringBuilder sb = new StringBuilder();
		
		sb.append(prefix);
		sb.append(word);
		sb.append(suffix);
		
		System.out.print(sb.toString());

	}
}

14.import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner sc = new Scanner(System.in);
		String s1 = sc.nextLine();
		String s2 = sc.nextLine();
		
		
		if(s1.contains(":")) s1 = s1.split(":")[1].trim();
		if(s2.contains(":")) s2 = s2.split(":")[1].trim();
		StringBuffer sb = new StringBuffer();
		int minLen = Math.min(s1.length(),s2.length());
		
		for(int i = 0;i < minLen;i++) {
		    sb.append(s1.charAt(i));
		    sb.append(s2.charAt(i));
		}
		if(s1.length() > minLen) {
		    sb.append(s1.substring(minLen));
		}
		if(s2.length() > minLen) {
		    sb.append(s2.substring(minLen));
		}
		
		
		System.out.print(sb.toString());

	}
}
