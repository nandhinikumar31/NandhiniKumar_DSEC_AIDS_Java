9.import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();

        String result = (a > 0 && b > 0 && c > 0 &&
                         a + b > c && a + c > b &&
                         b + c > a)
                        ? "Valid Triangle" : "Invalid Triangle";

        System.out.println(result);
        sc.close();
    }
}
