import java.io.*;

class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
	     Scanner sc = new Scanner(System.in);
		 int a[] = new int [5];
		 a[0] = 10;
		 a[1] = 20;
		 a[2] = 30;
		 a[3] = 40;
		 int n = 4;
		 
		 System.out.println("enter value:");
		 int value = sc.nextInt();
		 System.out.println("Enter index(0 to "+ n + "):");
		 int pos = sc.nextInt();
		 
		 if(pos < 0 || pos > n || n == a.length) {
		     System.out.println("Invalid Number");
		 } else {
		     for(int i = n;i > pos; i--) {
		         a[i] = a[i - 1];
		     }
		     a[pos] = value;
		     n++;
		     for(int i = 0; i < n;i++) {
		         System.out.println(a[i] + " ");
		     }
		     sc.close();
		 }

	}
}

5.import java.util.*;
import java.lang.*;
import java.io.*;

class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
	     
	     int a[] = {10,20,30,40,50};
		 int key = 40;
		 int low = 0;
		 int high = a.length - 1;
		 int pos = -1;
		 while(low == high) {
		     int mid = (low + high) / 2;
		     if(a[mid] == key) {
		         pos = mid;
		         break;
		     } else if (key > a[mid]) {
		         low = mid + 1;
		     } else {
		         high = mid - 1;
		     }
		     if(pos!= -1)
		     System.out.println("found at index" + pos);
		     else
		     System.out.println("not found");
		 }
	}
}

7.import java.util.*;
import java.lang.*;
import java.io.*;

class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
	     Scanner sc = new Scanner(System.in);
	     int a[] = new int[5];
	     System.out.println("enter 5 elements:");
	     for(int i = 0;i < 5;i++) {
	         a[i] = sc.nextInt();
	     }
	     int largest = a[0];
	     int second = Integer.MIN_VALUE;
	     for(int i = 1;i < 5; i++) {
	         if (a[i] > largest) {
	             second = largest;
	             largest = a[i];
	         } else if (a[i] > second && a[i] != largest) {
	             second = a[i];
	         }
	     }
	     if(second == Integer.MIN_VALUE)
	     System.out.println("second largest element not found");
	     else
	     System.out.println("second largest = " + second);
	} 
}

8.import java.util.*;

import java.io.*;

class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
	     int a[] = {0,5,0,3,8,0,2};
	     int j = 0;
	     for(int i = 0;i < a.length;i++) {
	         if(a[i] != 0) {
	             a[j] = a[i];
	             j++;
	         }
	     }
	     while(j < a.length) {
	         a[j] = 0;
	         j++;
	     }
	     System.out.println("array after moving zeros");
	     for(int i = 0; i < a.length;i++) {
	         System.out.println(a[i] + " ");
	     }
	} 
}

9.import java.util.*;
import java.lang.*;
import java.io.*;

class Main
{
	public static void main (String[] args) throws java.lang.Exception
	{
	     int a[] = {1,1,3,3,7,5};
	     int n = a.length;
	     
	     for(int i = 0;i < n;i++) {
	         boolean isDuplicate = false;
	         for(int j = 0;j < i;j++) {
	             if(a[i] == a[j]) {
	                 isDuplicate = true;
	                 break;
	             }
	         }
	         if(!isDuplicate) {
	             System.out.println(a[i] + " ");
	         }
	     }
	} 
}
