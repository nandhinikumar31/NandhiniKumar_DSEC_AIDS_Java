import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		Scanner sc = new Scanner(System.in);
		int n = sc.nextInt();
		int square = n * n;
		int sum = 0;
		while(square > 0){
		    int digit = square % 10;
		    sum = sum + digit;
		    square = square / 10;
		}
		if (sum == n) {
		    System.out.println("neon number");
		} else {
		    System.out.println("not a neon number");
		}
	}
}
