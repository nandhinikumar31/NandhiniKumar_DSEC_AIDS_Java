4.import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	static void mergeSort(int a[],int left,int right) {
	    if(left < right) {
	        int mid = (left + right) / 2;
	        mergeSort(a,left,mid);
	        mergeSort(a,mid + 1,right);
	        merge(a,left,mid,right);
	    }
	}
	static void merge(int a[],int left,int mid,int right) {
	    int temp[] = new int[right - left + 1];
	    int i = left;
	    int j = mid + 1;
	    int k = 0;
	    while(i <= mid && j <= right) {
	        if(a[i] <= a[j]) {
	            temp[k++] = a[i++];
	        } else {
	            temp[k++] = a[j++];
	        }
	    }
	    while(i <= mid) {
	        temp[k++] = a[i++];
	    }
	    while(j <= right) {
	        temp[k++] = a[j++];
	    }
	    for(i = left;i <= right;i++) {
	        a[i] = temp[i - left];
	    }
	}
	public  static void main(String[] args) {
	    Scanner sc  = new Scanner(System.in);
	    System.out.print("enter array size:");
	    int n = sc.nextInt();
	    int a[] = new int[n];
	    System.out.println("enter array elements:");
	    for(int i = 0;i < n;i++) {
	        a[i] = sc.nextInt();
	    }
	    mergeSort(a,0,n - 1);
	    System.out.println("sorted array:");
	    for(int i = 0;i < n;i++) {
	        System.out.print(a[i] + " ");
	    }
	}
}

                minElement = arr[i];
            }

10.import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        ArrayList<Integer> list = new ArrayList<>();
        while(sc.hasNextInt()){
            list.add(sc.nextInt());
        }
        
        if(list.isEmpty()) return;
        
        int[] arr = new int[list.size()];
        for(int i=0; i<list.size(); i++) arr[i] = list.get(i);

        // Leaders ah store panna
        ArrayList<Integer> leaders = new ArrayList<>();
        
        int maxFromRight = arr[arr.length - 1];
        leaders.add(maxFromRight);
        
        // Right la irunthu left porom - O(n)
        for (int i = arr.length - 2; i >= 0; i--) {
            if (arr[i] > maxFromRight) {
                maxFromRight = arr[i];
                leaders.add(maxFromRight);
            }
        }
        
        // Reverse pannanum because right la irunthu add pannom
        Collections.reverse(leaders);
        
        // Output print
        for (int i = 0; i < leaders.size(); i++) {
            System.out.print(leaders.get(i));
            if (i < leaders.size() - 1) System.out.print(" ");
        }
    }
}

        int totalSum = 0;
        for (int num : arr) totalSum += num;

        int leftSum = 0;
        int equilibriumIndex = -1;

        for (int i = 0; i < arr.length; i++) {
            // rightSum = totalSum - leftSum - arr[i]
            int rightSum = totalSum - leftSum - arr[i];
            
            if (leftSum == rightSum) {
                equilibriumIndex = i; // 0-based index
                break;
            }
            leftSum += arr[i];
        }

        if (equilibriumIndex != -1) {
            System.out.println("Equilibrium Index = " + equilibriumIndex);
        } else {
            System.out.println("No Equilibrium Index found");
        }
    }
}

        }

        int target = allNumbers.get(allNumbers.size() - 1);
        int[] arr = new int[allNumbers.size() - 1];

        // HashMap use panni O(n) la find pannalam
        HashSet<Integer> seen = new HashSet<>();
        boolean found = false;
        int num1 = 0, num2 = 0;

        for (int num : arr) {
            int complement = target - num;
            if (seen.contains(complement)) {
                num1 = complement;
                num2 = num;
                found = true;
                break;
            }
            seen.add(num);
        }

        if (found) {
            System.out.println("Pair = " + num1 + " " + num2);
        } else {
            System.out.println("No Pair Found");
        }
    }

        Scanner sc = new Scanner(System.in);
        
        ArrayList<Integer> list = new ArrayList<>();
        while(sc.hasNextInt()){
            list.add(sc.nextInt());
        }
        if(list.isEmpty()) return;
        
        int[] arr = new int[list.size()];
        for(int i=0; i<list.size(); i++) arr[i] = list.get(i);

        // Step 1: Candidate ah find pannu
        int count = 0;
        int candidate = 0;
        for (int num : arr) {
            if (count == 0) {
                candidate = num;
            }
            if (num == candidate) {
                count++;
            } else {
                count--;
            }
        }

        // Step 2: Verify pannu - candidate n/2 thadavai varutha nu
        int freq = 0;
        for (int num : arr) {
            if (num == candidate) freq++;
        }

        if (freq > arr.length / 2) {
            System.out.println("Majority Element = " + candidate);
        } else {
            System.out.println("No Majority Element");
        }
    }
}

            }
        }

        System.out.println("Length = " + maxLen);
    }

        }
    }

public class Main {

        while(sc.hasNextInt()){
            list.add(sc.nextInt());
        }
        if(list.isEmpty()) return;
        
        int[] arr = new int[list.size()];
        for(int i=0; i<list.size(); i++) arr[i] = list.get(i);
        
        int n = arr.length;

        // Step 1: Correct position la vechukkurom
        // 1 should be at index 0, 2 at index 1, etc.
        for (int i = 0; i < n; i++) {
            while (arr[i] > 0 && arr[i] <= n && arr[arr[i] - 1] != arr[i]) {
                // Swap arr[i] with arr[arr[i]-1]
                int temp = arr[arr[i] - 1];
                arr[arr[i] - 1] = arr[i];
                arr[i] = temp;
            }
        }

        // Step 2: Missing ah find pannu
        int missing = n + 1;
        for (int i = 0; i < n; i++) {
            if (arr[i] != i + 1) {
                missing = i + 1;
                break;
            }
        }

        System.out.println(missing);
    }
}
